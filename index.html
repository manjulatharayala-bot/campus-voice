const SHEET_NAME = "Complaints"; function setup() { const props = PropertiesService.getScriptProperties(); let id = props.getProperty("SPREADSHEET_ID"); if (!id) { const ss = SpreadsheetApp.create("Campus Voice Complaints"); id = ss.getId(); props.setProperty("SPREADSHEET_ID", id); const sheet = ss.getActiveSheet(); sheet.setName(SHEET_NAME); sheet.appendRow([ "ID", "Date", "Student Name", "College ID", "Category", "Location", "Description", "Priority", "Status", "Admin Reply" ]); } return "Setup complete"; } function doGet() { return HtmlService.createHtmlOutputFromFile("Index") .setTitle("Campus Voice"); } function getSheet() { const id = PropertiesService.getScriptProperties() .getProperty("SPREADSHEET_ID"); if (!id) throw new Error("Run setup() first."); return SpreadsheetApp.openById(id).getSheetByName(SHEET_NAME); } function submitComplaint(data) { if (!data || !data.name || !data.collegeId || !data.category || !data.description) { throw new Error("Please fill all required fields."); } const id = "CV-" + Date.now(); getSheet().appendRow([ id, new Date(), data.name, data.collegeId, data.category, data.location || "", data.description, data.priority || "Medium", "Submitted", "" ]); return "Complaint submitted successfully! Your ID: " + id; } function adminLogin(username, password) { const props = PropertiesService.getScriptProperties(); if ( username !== props.getProperty("ADMIN_USERNAME") || password !== props.getProperty("ADMIN_PASSWORD") ) { throw new Error("Invalid admin username or password."); } const token = Utilities.getUuid(); CacheService.getScriptCache().put( "admin_" + token, "valid", 21600 ); return token; } function checkAdmin(token) { return !!token && CacheService.getScriptCache().get("admin_" + token) === "valid"; } function getComplaints(token) { if (!checkAdmin(token)) { throw new Error("Please log in as admin."); } const rows = getSheet().getDataRange().getValues(); if (rows.length <= 1) return []; return rows.slice(1).map((r, i) => ({ row: i + 2, id: r[0], date: r[1] instanceof Date ? r[1].toLocaleString() : String(r[1]), name: r[2], collegeId: r[3], category: r[4], location: r[5], description: r[6], priority: r[7], status: r[8], reply: r[9] })); } function updateComplaint(token, row, status, reply) { if (!checkAdmin(token)) { throw new Error("Please log in as admin."); } const sheet = getSheet(); const rowNumber = Number(row); if (!Number.isInteger(rowNumber) || rowNumber < 2 || rowNumber > sheet.getLastRow()) { throw new Error("Invalid complaint."); } sheet.getRange(rowNumber, 9).setValue(status); sheet.getRange(rowNumber, 10).setValue(reply || ""); return "Complaint updated successfully."; }
